#include <bits/stdc++.h>
using namespace std;
int a[500005];
int main(){
	freopen("crystal.in","r",stdin);
	freopen("crystal.out","w",stdout);
	int n,m;
	cin>>n>>m;
	for(int i=1;i<=n;i++){
		cin>>a[i];
	}
	int ans=0;
	for(int i=1;i<=n;i++){
		ans+=a[i];
	}
	int bit=0;
	for(int i=1;i<=n;i++){
		if(a[i]<(ans+m)/n){
			bit++;
		}
	}
	cout<<bit;
	return 0;
}
